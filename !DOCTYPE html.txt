        function loadData() {

            keys.forEach((key, idx) => {
                const data = localStorage.getItem(key);
                if (data) {
                    if (idx === 0) kuryeler = JSON.parse(data);
                    if (idx === 1) motorlar = JSON.parse(data);
                    if (idx === 2) bakimKayitlari = JSON.parse(data);
                    if (idx === 3) geriBildirimler = JSON.parse(data);
                }
            });

        }

            localStorage.setItem('mototakip_geribildirimler', JSON.stringify(geriBildirimler));

                return;

        function saveGeriBildirim() {

                } else if (durum === 'degistirildi') {
                    yeniParcaSayisi++;
                    durumOzeti.push(`${kategoriAdi}: Yeni parça`);
                } else if (durum === 'orta') {
                    durumOzeti.push(`${kategoriAdi}: Dikkat`);
                }
            });

            const geriBildirim = {
                id: Date.now(),
                plaka: motor.plaka,
                model: motor.model,
                kuryeId: kurye.id,
                kuryeIsim: kurye.isim,
                kuryeTelefon: kurye.telefon,
                tarih: new Date().toISOString(),
                durumlar: {...currentGeriBildirim},
                durumOzeti: durumOzeti,
                sorunSayisi: sorunSayisi,
                yeniParcaSayisi: yeniParcaSayisi,
                not: not,
                okundu: false
            };

            geriBildirimler.push(geriBildirim);
            saveData();

            currentGeriBildirim = {};
            document.getElementById('geri-bildirim-not').value = '';
            document.querySelectorAll('.feedback-btn').forEach(btn => {
                btn.classList.remove('ring-4', 'ring-offset-2', 'scale-110', 'shadow-lg');
                btn.style.opacity = '1';
            });

            let mesaj = `✅ Geri bildirim gönderildi!\n\n📋 Özet:\n`;
            if (sorunSayisi > 0) mesaj += `⚠️ ${sorunSayisi} sorunlu parça\n`;
            if (yeniParcaSayisi > 0) mesaj += `🔄 ${yeniParcaSayisi} yeni parça\n`;
            if (durumOzeti.length === 0) mesaj += `✓ Her şey yolunda\n`;
            
            alert(mesaj);
        }

        function renderGeriBildirimler(filter = 'tumu') {

        function bakimYaptir() {
            const yer = document.getElementById('secili-bakim-yeri').value;
            const tip = document.getElementById('secili-bakim-tipi').value;
            const km = parseInt(document.getElementById('bakim-km-input').value);

            if (!yer || !km) {
                alert('Tüm alanları doldurun!');
                return;
            }

            const kurye = kuryeler.find(k => k.id === currentUser.id);
            const motor = motorlar.find(m => m.plaka === kurye.plaka);

            motor.sonBakimKm = km;
            motor.sonBakimTarihi = new Date().toISOString();

            bakimKayitlari.push({

                plaka: motor.plaka,
                model: motor.model,
                tip: tip,
                km: km,
                yer: yer,
                kuryeIsim: kurye.isim,
                kuryeId: kurye.id,
                telefon: kurye.telefon,
                tarih: new Date().toISOString()
            });

            saveData();
            document.getElementById('bakim-km-input').value = '';
            document.querySelectorAll('.bakim-yeri-btn').forEach(btn => btn.classList.remove('ring-4', 'ring-brand-orange', 'bg-brand-lightYellow'));

            alert(`✅ ${tipText} kaydedildi!\n\n📍 ${yer}\n📊 ${km.toLocaleString()} km`);

        }

                const geriBildirimSayisi = geriBildirimler.filter(g => g.kuryeId === k.id).length;
                kuryeHtml += `
                    <div class="flex justify-between items-center p-3 bg-brand-lightYellow/30 rounded-lg">
                        <div>
                            <span class="font-bold text-brand-dark">${k.isim}</span>
                            ${k.telefon ? `<span class="text-sm text-gray-500 ml-2">📞 ${k.telefon}</span>` : ''}
                        </div>
                        <div>
                            <span class="px-3 py-1 bg-brand-orange text-white rounded-full text-sm font-bold mr-2">${bakimSayisi} bakım</span>
                            <span class="px-3 py-1 bg-brand-green text-white rounded-full text-sm font-bold">${geriBildirimSayisi} bildirim</span>
                        </div>
                    </div>
                `;
            });
            document.getElementById('rapor-kurye-listesi').innerHTML = kuryeHtml || '<p class="text-gray-500">Kurye yok</p>';

            const sonGeriBildirimler = [...geriBildirimler].sort((a,b) => new Date(b.tarih) - new Date(a.tarih)).slice(0, 5);
            document.getElementById('rapor-geri-bildirimler').innerHTML = sonGeriBildirimler.map(g => {
                let durumText = '✓ Normal';
                if (g.sorunSayisi > 0) durumText = `⚠️ ${g.sorunSayisi} sorun`;
                else if (g.yeniParcaSayisi > 0) durumText = `🔄 ${g.yeniParcaSayisi} yeni`;
                
                return `
                    <div class="flex justify-between items-center p-2 bg-gray-50 rounded text-sm">
                        <span class="font-bold ${g.sorunSayisi > 0 ? 'text-red-500' : (g.yeniParcaSayisi > 0 ? 'text-brand-blue' : 'text-green-500')}">
                            ${g.plaka} - ${durumText}
                        </span>
                        <span class="text-gray-600">${g.kuryeIsim}</span>
                        <span class="text-gray-500">${new Date(g.tarih).toLocaleDateString('tr-TR')}</span>
                    </div>
                `;
            }).join('') || '<p class="text-gray-500">Geri bildirim yok</p>';

        }

        function saveKurye() {
            const id = document.getElementById('kurye-id').value;
            const isim = document.getElementById('kurye-modal-isim').value.trim();
            const telefon = document.getElementById('kurye-modal-telefon').value.trim();
            const plaka = document.getElementById('kurye-modal-plaka').value.trim().toUpperCase();

            if (!isim || !plaka) {
                alert('İsim ve plaka zorunludur!');
                return;
            }

            let motor = motorlar.find(m => m.plaka === plaka);
            if (!motor) {
                motor = {
                    id: Date.now(),
                    plaka: plaka,
                    model: 'Bilinmiyor',
                    yil: new Date().getFullYear(),
                    km: 0
                };
                motorlar.push(motor);
            }

            if (id) {
                const k = kuryeler.find(x => x.id == id);
                if (k) {
                    k.isim = isim;
                    k.telefon = telefon;
                    k.plaka = plaka;
                }
            } else {
                kuryeler.push({
                    id: Date.now(),
                    isim: isim,
                    telefon: telefon,
                    plaka: plaka,
                    durum: 'aktif',
                    ilkGiris: new Date().toISOString()
                });
            }

            saveData();
            closeModal('kurye');
            renderKuryeList();
            generateReport();
        }

            renderGeriBildirimler();
            generateReport();
        }

// DURUM HESAPLA
// ========================
function calcDurum(arac) {
  const kalanKm = (arac.sonBakim + arac.esik) - arac.anlikKm;
  const pct = Math.max(0, Math.min(100, (arac.anlikKm - arac.sonBakim) / arac.esik * 100));
  let durum = 'ok', label = 'Normal';
  if (kalanKm <= 0) { durum = 'danger'; label = 'Gecikti!'; }
  else if (kalanKm <= 1000) { durum = 'warn'; label = 'Yaklasıyor'; }
  return { kalanKm, pct, durum, label, hedefKm: arac.sonBakim + arac.esik };
}

function aracEkle() {
  let valid = true;
  const plaka = document.getElementById('inp-plaka').value.trim().toUpperCase();
  const kurye = document.getElementById('inp-kurye').value.trim();
  const marka = document.getElementById('inp-marka').value.trim();
  const anlikKm = parseInt(document.getElementById('inp-anlik').value);
  const esik = parseInt(document.getElementById('inp-esik').value);
  const sonBakim = parseInt(document.getElementById('inp-sonbakim').value);

  clearErrors('plaka','kurye','anlik','esik','sonbakim');

  if (!plaka) { setError('plaka', true); valid = false; }
  if (!kurye) { setError('kurye', true); valid = false; }
  if (isNaN(anlikKm) || anlikKm < 0) { setError('anlik', true); valid = false; }
  if (isNaN(esik) || esik < 100) { setError('esik', true); valid = false; }
  if (isNaN(sonBakim) || sonBakim < 0) { setError('sonbakim', true); valid = false; }

  if (!valid) return;

  if (araclar.find(a => a.plaka === plaka)) {
    showToast('Hata', 'Bu plaka zaten kayitli!', 'danger');
    return;
  }

  araclar.push({ plaka, kurye, marka, anlikKm, esik, sonBakim });
  saveData();
  updateAll();
  formTemizle();
  showToast('Basarili!', plaka + ' aracı sisteme eklendi.', 'ok');

  const d = calcDurum({ anlikKm, esik, sonBakim });
  if (d.durum === 'danger') showToast('KRITIK UYARI!', plaka + ' aracının bakimi gecikti!', 'danger');
  else if (d.durum === 'warn') showToast('Uyari', plaka + ' aracinin bakimina ' + d.kalanKm + ' km kaldi.', 'warn');
}

function aracSil(i) {
  if (!confirm('Bu araci silmek istediginize emin misiniz?')) return;
  const plaka = araclar[i].plaka;
  araclar.splice(i, 1);
  saveData();
  updateAll();
  renderCharts();
  showToast('Silindi', plaka + ' aracı kaldırıldı.', 'warn');
}

function formTemizle() {
  ['inp-plaka','inp-kurye','inp-marka','inp-anlik','inp-esik','inp-sonbakim'].forEach(id => document.getElementById(id).value = '');
  clearErrors('plaka','kurye','anlik','esik','sonbakim');
}

function kmHesaplaVeKaydet() {
  const plaka = document.getElementById('km-plaka').value;
  const yeniKm = parseInt(document.getElementById('km-yeni').value);
  const fgYeni = document.getElementById('fg-yeniKm');

  fgYeni.classList.remove('has-error');

  if (!plaka) { showToast('Hata', 'Lutfen bir arac secin.', 'danger'); return; }
  if (isNaN(yeniKm) || yeniKm < 0) { fgYeni.classList.add('has-error'); return; }

  const idx = araclar.findIndex(x => x.plaka === plaka);
  if (idx === -1) return;

  const a = araclar[idx];
  if (yeniKm < a.anlikKm) {
    showToast('Hata', 'Yeni KM, mevcut kayitli KM degeri olan ' + a.anlikKm.toLocaleString() + "'den kucuk olamaz.", 'danger');
    return;
  }

  araclar[idx].anlikKm = yeniKm;
  saveData();
  updateAll();

  const d = calcDurum(araclar[idx]);
  showKmResult('live', d, yeniKm);

  showToast('Guncellendi!', plaka + ' - yeni KM: ' + yeniKm.toLocaleString(), 'ok');
  if (d.durum === 'danger') showToast('KRITIK!', 'Bakim tarihi gecti! Hemen bakim yapılması gerekiyor.', 'danger');
  else if (d.durum === 'warn') showToast('Uyari!', 'Bakima ' + d.kalanKm.toLocaleString() + ' km kaldi.', 'warn');
  renderCharts();
}

  if (!plaka) return;

function showKmResult(prefix, d, km) {
  const result = document.getElementById(prefix === 'live' ? 'kmLiveResult' : 'modalResult');
  result.classList.add('show');
  const pfill = document.getElementById(prefix === 'live' ? 'liveProgressFill' : 'modalProgressFill');
  const ppct = document.getElementById(prefix === 'live' ? 'liveProgressPct' : 'modalProgressPct');
  const pLabel = document.getElementById(prefix === 'live' ? 'liveProgressLabel' : 'modalProgressLabel');
  const kalan = document.getElementById(prefix === 'live' ? 'liveKalanKm' : 'modalKalanKm');
  const durum = document.getElementById(prefix === 'live' ? 'liveDurum' : 'modalDurum');

  pfill.style.width = Math.min(100, d.pct) + '%';
  pfill.className = 'progress-fill ' + d.durum;
  ppct.textContent = Math.round(d.pct) + '%';
  pLabel.textContent = 'Bakim: ' + (d.kalanKm > 0 ? d.kalanKm.toLocaleString() + ' km kaldi' : 'GECIKTI!');
  kalan.textContent = d.kalanKm > 0 ? '+' + d.kalanKm.toLocaleString() + ' km' : d.kalanKm.toLocaleString() + ' km';
  kalan.style.color = 'var(--' + (d.durum === 'ok' ? 'ok' : d.durum === 'warn' ? 'warn' : 'danger') + ')';
  durum.innerHTML = '<span class="badge ' + d.durum + '">' + d.label + '</span>';

  if (prefix === 'live') {
    document.getElementById('liveHedefKm').textContent = d.hedefKm.toLocaleString() + ' km';
  }
}

let chartDurum, chartBar;
function renderCharts() {
  const cs = getComputedStyle(document.documentElement);
  const ok = cs.getPropertyValue('--ok').trim();
  const warn = cs.getPropertyValue('--warn').trim();
  const danger = cs.getPropertyValue('--danger').trim();
  const text2 = cs.getPropertyValue('--text2').trim();

  let okN = 0, warnN = 0, dangerN = 0;
  araclar.forEach(a => {
    const d = calcDurum(a);
    if (d.durum === 'ok') okN++;
    else if (d.durum === 'warn') warnN++;
    else dangerN++;
  });

  // DONUT
  if (chartDurum) chartDurum.destroy();
  const ctx1 = document.getElementById('chartDurum').getContext('2d');
  chartDurum = new Chart(ctx1, {
    type: 'doughnut',
    data: {
      labels: ['Normal', 'Yaklasıyor', 'Kritik'],
      datasets: [{ data: [okN, warnN, dangerN], backgroundColor: [ok, warn, danger], borderWidth: 0, hoverOffset: 8 }]
    },
    options: {
      responsive: true,
      cutout: '68%',
      plugins: {
        legend: { position: 'bottom', labels: { color: text2, font: { family: 'Space Grotesk', size: 12 }, padding: 16 } }
      }
    }
  });

  // BAR
  if (chartBar) chartBar.destroy();
  const ctx2 = document.getElementById('chartBar').getContext('2d');
  const labels = araclar.map(a => a.plaka);
  const kalanKmArr = araclar.map(a => Math.max(0, calcDurum(a).kalanKm));
  const bgColors = araclar.map(a => {
    const d = calcDurum(a);
    return d.durum === 'ok' ? ok : d.durum === 'warn' ? warn : danger;
  });
  chartBar = new Chart(ctx2, {
    type: 'bar',
    data: {
      labels: labels.length ? labels : ['Veri Yok'],
      datasets: [{ label: 'Kalan KM', data: kalanKmArr.length ? kalanKmArr : [0], backgroundColor: bgColors.length ? bgColors : [text2], borderRadius: 6, borderSkipped: false }]
    },
    options: {
      responsive: true,
      plugins: { legend: { display: false } },
      scales: {
        x: { ticks: { color: text2, font: { family: 'JetBrains Mono', size: 11 } }, grid: { display: false } },
        y: { ticks: { color: text2 }, grid: { color: 'rgba(255,255,255,0.05)' } }
      }
    }
  });
}

function exportPDF() {
  const { jsPDF } = window.jspdf;
  const doc = new jsPDF();
  doc.setFontSize(18); doc.text('MotoTakip - Tam Rapor', 14, 20);
  doc.setFontSize(10); doc.setTextColor(120); doc.text(new Date().toLocaleString('tr-TR'), 14, 28);
  doc.setTextColor(0);
  let y = 40;
  doc.setFontSize(11);
  doc.text('Plaka', 14, y); doc.text('Kurye', 50, y); doc.text('Anlık KM', 100, y); doc.text('Kalan KM', 140, y); doc.text('Durum', 175, y);
  y += 6; doc.line(14, y, 196, y); y += 6;
  doc.setFontSize(10);
  araclar.forEach(a => {
    const d = calcDurum(a);
    doc.text(a.plaka, 14, y);
    doc.text(a.kurye, 50, y);
    doc.text(a.anlikKm.toLocaleString() + ' km', 100, y);
    doc.text(d.kalanKm > 0 ? d.kalanKm.toLocaleString() + ' km' : 'GECIKTI!', 140, y);
    doc.text(d.label, 175, y);
    y += 8;
    if (y > 270) { doc.addPage(); y = 20; }
  });
  if (araclar.length === 0) doc.text('Kayitli arac bulunmuyor.', 14, y);
  doc.save('mototakip-rapor-' + new Date().toISOString().slice(0,10) + '.pdf');
  showToast('PDF', 'Rapor indirildi.', 'ok');
}

  if (kritikler.length === 0) { doc.text('Kritik arac bulunmuyor.', 14, y); }
  else {
    kritikler.forEach(a => {
      const d = calcDurum(a);
      doc.setTextColor(d.durum === 'danger' ? [255,0,0] : [200,120,0]);
      doc.text(a.plaka + ' | ' + a.kurye + ' | Kalan: ' + (d.kalanKm > 0 ? d.kalanKm.toLocaleString() + ' km' : 'GECIKTI!') + ' | ' + d.label, 14, y);
      y += 8;
    });
  }
  doc.save('mototakip-kritik-' + new Date().toISOString().slice(0,10) + '.pdf');
  showToast('PDF', 'Kritik rapor indirildi.', 'warn');
}

function exportCSV() {
  let csv = 'Plaka,Kurye,Marka,Anlık KM,Son Bakim KM,Bakim Araligi,Kalan KM,Durum\n';
  araclar.forEach(a => {
    const d = calcDurum(a);
    csv += [a.plaka, a.kurye, a.marka || '', a.anlikKm, a.sonBakim, a.esik, d.kalanKm, d.label].join(',') + '\n';
  });
  const blob = new Blob(['\uFEFF' + csv], { type: 'text/csv;charset=utf-8;' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url; a.download = 'mototakip-' + new Date().toISOString().slice(0,10) + '.csv';
  a.click();
  URL.revokeObjectURL(url);
  showToast('CSV', 'Veri disa aktarildi.', 'ok');
}
